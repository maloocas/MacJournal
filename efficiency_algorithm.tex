\section*{LMKPI Daily Efficiency Algorithm}

\subsection*{Inputs}
\begin{itemize}
  \item $C = \text{proDone} + \text{perDone}$ — total tasks completed
  \item $T = \text{proTotal} + \text{perTotal}$ — total tasks assigned
  \item $s$ — sleep hours
  \item $m$ — social minutes
  \item $r$ — reading pages
  \item $\text{meal}_i \in \{\text{Healthy, Fancy, Standard, Junk, Skipped}\},\; i = 1,2,3$
\end{itemize}

\subsection*{Constants}
\[
\text{readingTarget} = 50,\quad \text{socialWeight} = 0.25,\quad \text{sleepMin} = 7.0,\quad \text{sleepMax} = 9.0,\quad \text{sleepPenalty} = 6.0
\]

\subsection*{TDI (Tasks Done Index)}
\[
\text{TDI} = \begin{cases}
\displaystyle\frac{C}{T} \times 100, & T > 0 \\[8pt]
0, & T = 0
\end{cases}
\]

\subsection*{Diet Score}
\[
d(m) = \begin{cases}
+5, & m = \text{Healthy} \\
-5, & m \in \{\text{Junk, Skipped}\} \\
0, & \text{otherwise}
\end{cases}
\qquad
D = d(\text{meal}_1) + d(\text{meal}_2) + d(\text{meal}_3)
\]

\subsection*{Sleep Score}
\[
S = \begin{cases}
+10, & 7.0 \le s \le 9.0 \\
-10, & s < 6.0 \\
0, & \text{otherwise}
\end{cases}
\]

\subsection*{Reading Contribution}
\[
R = \min(r, 50) \times 0.4
\]

\subsection*{Efficiency (clamped 0–100)}
\[
\boxed{E = \operatorname{clamp}_{[0,100]}\Bigl(50 + 0.3 \cdot \text{TDI} + R + D + S - 0.25 \cdot m\Bigr)}
\]

\subsection*{Component Ranges}
\begin{center}
\begin{tabular}{lrl}
\textbf{Component} & \textbf{Contribution} & \textbf{Range} \\ \hline
Base & $+50$ & fixed \\
TDI $\times 0.3$ & & $[0, 30]$ \\
Reading & $\min(r,50) \times 0.4$ & $[0, 20]$ \\
Diet & $\pm 5$ per meal & $[-15, 15]$ \\
Sleep & ternary & $\{-10, 0, 10\}$ \\
Social & $-0.25$ per minute & $(-\infty, 0]$ \\ \hline
\textbf{Total} & \textbf{clamped} & $\mathbf{[0, 100]}$
\end{tabular}
\end{center}
